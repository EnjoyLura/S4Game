/** Three chapter scenes, ten-stage routes and a persistent reward / battle dock. */
import { Graphics, Node, ScrollView, Sprite, UIOpacity, Vec2, director } from 'cc';
import { LO } from '../../config/GameConfig';
import { CAMPAIGN_CHAPTERS, CAMPAIGN_STAGES, campaignDifficultyOpen, campaignLevelOpen } from '../../config/Campaign';
import { C, CA, N, WY } from '../UIKit';
import { artSprite } from '../Ux';
import { COMIC, comicArt, comicButton, comicButtonControl, comicMark, comicPlate, comicSurface } from '../ComicUI';
import { mainLabel, PageCtx, toast, mainLayout, MAIN_RESOURCE_H, pageScroll } from '../PageKit';
import { CHEST_COND, claimAllChests, starKey, squadList } from '../../core/GameData';
import { DIFFICULTIES, getChallengeDef, starChestRewards } from '../../config/LevelDifficulty';
import { stageLootPreview } from '../LevelLootPreview';
import { loadSave } from '../../core/SaveData';
import { flow } from '../../core/Flow';
import { openSquadPanel } from '../SquadPanel';
import { bindButton } from '../ButtonFeedback';
import { refreshUI } from '../UIMotion';
import { enterBattle, enteringBattle } from '../BattleEntry';
import { warmBattlefield } from '../BattleEntryWarmup';
import { deferUIWarmup } from '../IdleWarmup';
import { showRewardPopup } from '../RewardPopup';

const LEVELS = CAMPAIGN_STAGES;
const DIFFS = ['普通', '精英', '地狱'];
const DIFFICULTY_ART = ['comic_difficulty_shield_v1', 'comic_swords_v1', 'comic_difficulty_skull_v1'];
const CHAPTER_NUMERALS = ['一', '二', '三'];
const st = { i: 0, d: 0, rewardTab: 0, initialized: false,
  selected: [0, 0, 0], upperOffset: [0, 0, 0] };

function levelArt(parent: Node, x: number, y: number, w: number, h: number, key: string): Node {
  return comicArt(parent, x, y, w, h, 'levels_comic_v1/' + key);
}
function tap(node: Node, cb: () => void): void {
  bindButton(node, cb);
}
function star(parent: Node, x: number, y: number, size: number, earned: boolean): void {
  const art = artSprite(parent, x, y, size, size, 'levels_comic_v1/comic_star_v1', {
    sliced: false, onLoaded: n => { if (!earned) n.getComponent(Sprite)!.color = C(COMIC.muted); },
  });
  if (!earned) art.addComponent(UIOpacity).opacity = 95;
}
/** Short screens scroll the whole chapter area; a drag must never select a stop. */
function routeTap(node: Node, cb: () => void): void {
  bindButton(node, cb, { dragThreshold: 7 });
}

export function buildLevels(ctx: PageCtx): void {
  st.rewardTab = 0; // Enter the level page on the left-hand star chest tab.
  if (!st.initialized) {
    st.initialized = true;
    const stars = loadSave().stars;
    CAMPAIGN_CHAPTERS.forEach((_, chapter) => {
      const stages = LEVELS.slice(chapter * 10, chapter * 10 + 10);
      const next = stages.find(stage => campaignLevelOpen(stage.id, stars) && !(stars[stage.id] > 0));
      st.selected[chapter] = next ? next.stage - 1 : stages.some(stage => stars[stage.id] > 0) ? 9 : 0;
    });
    const next = LEVELS.find(stage => campaignLevelOpen(stage.id, stars) && !(stars[stage.id] > 0));
    st.i = next ? next.order : LEVELS.length - 1;
  }
  // Consume only the battle-exit request; ordinary lobby switches keep their remembered selection.
  const focus = flow.levelReturn;
  if (focus) {
    flow.levelReturn = null;
    const index = LEVELS.findIndex(level => level.id === focus.id);
    if (index >= 0) {
      st.i = index;
      st.d = Math.max(0, Math.min(DIFFS.length - 1, Math.floor(focus.diff)));
      while (st.d > 0 && !diffUnlocked(st.i, st.d)) st.d--;
      st.selected[Math.floor(index / 10)] = index % 10;
      st.upperOffset[Math.floor(index / 10)] = 0;
    }
  }
  const content = N('levelLobby', ctx.screens, 0, 0, 750, LO.half * 2);
  let claiming = false;
  function levelOpen(i: number): boolean {
    return !!LEVELS[i] && campaignLevelOpen(LEVELS[i].id, loadSave().stars);
  }
  function diffUnlocked(i: number, d: number): boolean {
    return !!LEVELS[i] && campaignDifficultyOpen(LEVELS[i].id, d, loadSave().stars);
  }
  function selectLevel(i: number): void {
    st.i = i;
    st.selected[Math.floor(i / 10)] = i % 10;
    while (st.d > 0 && !diffUnlocked(st.i, st.d)) st.d--;
    render();
  }
  function render(): void {
    refreshUI(content, () => {
      const sv = loadSave(), lv = LEVELS[st.i], open = levelOpen(st.i);
      const chapterIndex = lv.chapter - 1, chapter = CAMPAIGN_CHAPTERS[chapterIndex];
      const challenge = getChallengeDef(lv.id, st.d), chestRewards = starChestRewards(lv.id, st.d);
      const key = starKey(lv.id, st.d), stars = sv.stars[key] || 0, claimed = sv.chestClaimed[key] || 0;
      const { headerTop, bottom } = mainLayout(), height = LO.half * 2;
      const titleTop = headerTop + MAIN_RESOURCE_H + 14;
      const actionH = 94, actionTop = bottom - actionH;
      // The reward content owns real vertical space instead of being squeezed under an expanding diorama.
      const rewardH = Math.min(460, Math.max(390, height * .30));
      const rewardTop = actionTop - rewardH - 14;

      const upper = N('levelUpperScene', content, 0, WY(0, rewardTop), 750, rewardTop);
      const terrain = artSprite(upper, 0, 0, 750, rewardTop,
        'scenes/bg_levels_' + chapter.art + '_comic_v2', { sliced: false, cover: true });
      terrain.name = 'levelTerrain:chapter' + chapter.number;
      // Keep the chapter route readable on short screens instead of shrinking touch targets.
      const upperViewH = Math.max(1, rewardTop - 12 - titleTop), upperH = Math.max(500, upperViewH);
      const body = pageScroll(content, 'chapterRoute', titleTop, rewardTop - 12, upperH, 718);
      const scroll = body.parent!.getComponent(ScrollView)!;
      scroll.inertia = true; scroll.brake = .65; scroll.cancelInnerEvents = true;
      scroll.elastic = false;
      const remember = (): void => {
        st.upperOffset[chapterIndex] = Math.min(Math.max(0, upperH - upperViewH), Math.max(0, scroll.getScrollOffset().y));
      };
      body.parent!.on('scrolling', remember);
      body.parent!.on('scroll-ended', remember);
      scroll.scrollToOffset(new Vec2(0, Math.min(st.upperOffset[chapterIndex], upperH - upperViewH)), 0);
      const by = (top: number, h = 0): number => -top - h / 2;
      const chapterStages = LEVELS.slice(chapterIndex * 10, chapterIndex * 10 + 10);
      const cleared = chapterStages.filter(stage => (sv.stars[stage.id] || 0) > 0).length;
      mainLabel(body, -147, by(0, 40), '第' + CHAPTER_NUMERALS[chapterIndex] + '章 · ' + chapter.name, {
        size: 32, color: COMIC.cream, bold: true, w: 388, h: 40, align: 'left', outline: COMIC.ink, outlineW: 3,
      }).name = 'levelTitle';
      mainLabel(body, 237, by(4, 32), '已通关 ' + cleared + '/10', {
        size: 22, color: COMIC.cream, w: 208, h: 32, outline: COMIC.ink, outlineW: 2,
      });
      CAMPAIGN_CHAPTERS.forEach((item, index) => {
        const active = chapterIndex === index, unlocked = levelOpen(index * 10);
        const tab = comicPlate(body, (index - 1) * 232, by(52, 54), 222, 54, active ? 'gold' : 'dark',
          { name: 'chapterSelect:' + item.number });
        mainLabel(tab, unlocked ? 0 : -10, 0, '第' + CHAPTER_NUMERALS[index] + '章  ' + item.name, {
          size: 23, bold: true, color: active ? COMIC.text : unlocked ? COMIC.cream : COMIC.muted, w: 194, h: 36,
        });
        if (!unlocked) comicMark(tab, 91, 0, 'lock', 16, active ? COMIC.text : COMIC.muted);
        routeTap(tab, () => { if (!active) selectLevel(index * 10 + st.selected[index]); });
      });

      const tabH = 50;
      const difficultyBar = N('levelDifficultyBar', body, 0, by(118, tabH), 686, tabH);
      DIFFS.forEach((name, d) => {
        const unlocked = diffUnlocked(st.i, d), active = st.d === d;
        const slot = comicPlate(difficultyBar, (d - 1) * 232, 0, 222, tabH, active ? 'gold' : 'dark',
          { name: 'levelDifficulty:' + d });
        levelArt(slot, -65, 0, 34, 34, DIFFICULTY_ART[d]);
        mainLabel(slot, 7, 0, name, {
          size: 25, color: active ? COMIC.text : unlocked ? COMIC.cream : COMIC.muted,
          bold: true, w: 94, h: 40,
        });
        if (!unlocked) comicMark(slot, 80, 0, 'lock', 20, COMIC.muted);
        routeTap(slot, () => {
          if (!unlocked) { toast(d === 0 ? '通关上一关解锁' : '通关上一难度解锁'); return; }
          if (st.d !== d) { st.d = d; render(); }
        });
      });

      const routeH = upperH - 266;
      const route = N('chapterStageRoute', body, 0, by(182, routeH), 704, routeH);
      const rowGap = Math.min(210, routeH - 122), r = routeH > 280 ? 35 : 31;
      const points = chapterStages.map((_, i) => ({ x: (i < 5 ? i - 2 : 7 - i) * 130,
        y: (i < 5 ? .5 : -.5) * rowGap }));
      const path = N('chapterRoutePath', route).addComponent(Graphics);
      for (let i = 1; i < points.length; i++) {
        const a = points[i - 1], b = points[i];
        const segment = (): void => {
          path.moveTo(a.x, a.y);
          if (i === 5) path.bezierCurveTo(354, a.y, 354, b.y, b.x, b.y);
          else path.lineTo(b.x, b.y);
          path.stroke();
        };
        path.lineWidth = 11; path.strokeColor = CA(COMIC.darkShade, .8);
        segment();
        path.lineWidth = 5;
        path.strokeColor = C((sv.stars[chapterStages[i - 1].id] || 0) > 0 ? COMIC.gold : COMIC.paperShade);
        segment();
      }
      chapterStages.forEach((level, i) => {
        const active = lv.id === level.id, unlocked = diffUnlocked(level.order, st.d);
        const earned = sv.stars[starKey(level.id, st.d)] || 0, p = points[i];
        const item = N('levelSelect:' + level.id, route, p.x, p.y, 108, 112);
        const g = item.addComponent(Graphics);
        if (active) {
          g.strokeColor = C(COMIC.goldLight); g.lineWidth = 3;
          g.circle(0, 0, r + 6); g.stroke();
        }
        g.fillColor = C(active ? COMIC.gold : unlocked ? COMIC.dark : COMIC.mutedShade);
        g.strokeColor = C(COMIC.ink); g.lineWidth = 3;
        g.circle(0, 0, r); g.fill(); g.stroke();
        g.strokeColor = C(active ? COMIC.goldLight : unlocked ? COMIC.darkLight : COMIC.muted);
        g.lineWidth = 2; g.arc(0, 0, r - 4, .35, Math.PI - .35, false); g.stroke();
        mainLabel(item, 0, 1, String(level.stage), {
          size: 29, bold: true, color: active ? COMIC.text : COMIC.cream, w: 62, h: 40,
        });
        if (!unlocked) comicMark(item, r - 2, r - 2, 'lock', 17, COMIC.cream);
        else if (earned > 0) comicMark(item, r - 2, r - 2, 'check', 18, COMIC.greenLight);
        for (let s = 0; s < 3; s++) star(item, (s - 1) * 21, -r - 16, 19, s < earned);
        if (level.stage === 10) comicMark(item, -r + 2, r - 2, 'upgrade', 18, COMIC.goldLight);
        routeTap(item, () => { if (!active) selectLevel(level.order); });
      });
      const summary = comicPlate(body, 0, by(upperH - 78, 74), 686, 74, 'dark', { name: 'selectedStage' });
      mainLabel(summary, 0, 15, lv.id + '  ' + lv.name + (lv.stage === 10 ? ' · 章末' : ''), {
        size: 25, bold: true, color: COMIC.cream, w: 652, h: 34,
      });
      mainLabel(summary, 0, -18, open ? challenge.waves.length + ' 波 · ' + DIFFICULTIES[st.d].formation
        : '通关 ' + LEVELS[st.i - 1].id + ' 普通难度后解锁', {
        size: 18, color: open ? COMIC.paperShade : COMIC.goldLight, w: 652, h: 28,
      });

      const rewards = comicPlate(content, 0, WY(rewardTop, rewardH), 750, rewardH, 'paper',
        { name: 'rewardShelf', cut: 0, edge: false, highlight: false });
      const ry = (top: number): number => rewardH / 2 - top;
      levelArt(rewards, -311, ry(28), 32, 32, 'comic_chest_closed_v1');
      mainLabel(rewards, -180, ry(28), '关卡奖励', {
        size: 27, color: COMIC.text, bold: true, w: 220, h: 36, align: 'left',
      });
      mainLabel(rewards, 147, ry(28), stars + '/3 星', { size: 22, color: COMIC.sub, w: 88, h: 30 });
      for (let i = 0; i < 3; i++) star(rewards, 218 + i * 36, ry(27), 31, i < stars);
      comicSurface(rewards, 0, ry(80), 652, 44, 'paper', { name: 'rewardTabs', radius: 12, fill: '#EAD9B7' });
      ['星级宝箱' + (stars > claimed ? ' · 可领取' : ''), '关卡掉落'].forEach((text, index) => {
        const active = st.rewardTab === index;
        const tab = comicPlate(rewards, (index - .5) * 326, ry(80), 320, 40, active ? 'gold' : 'paper', {
          name: 'rewardTab:' + index, edge: false, highlight: false, fill: active ? COMIC.gold : '#EAD9B7',
        });
        mainLabel(tab, 0, 0, text, { size: 21, color: active ? COMIC.text : COMIC.sub, bold: true, w: 302, h: 30 });
        tap(tab, () => { if (st.rewardTab !== index) { st.rewardTab = index; render(); } });
      });
      const bodyTop = 114, bodyH = rewardH - bodyTop - 14;
      if (st.rewardTab === 1) {
        const preview = N('stageLootPreview', rewards, 0, ry(bodyTop + bodyH / 2), 660, bodyH);
        stageLootPreview(preview, bodyH, challenge, ctx.screens.parent!);
      } else {
        for (let t = 0; t < 3; t++) {
          const tier = t + 1, canClaim = stars >= tier && claimed < tier, done = claimed >= tier;
          const spot = N('rewardTier' + tier, rewards, (t - 1) * 222, ry(bodyTop + bodyH / 2), 210, bodyH);
          const sy = (top: number): number => bodyH / 2 - top;
          const condition = mainLabel(spot, 0, sy(22), CHEST_COND[t], { size: 20, color: COMIC.sub, w: 208, h: 28 });
          condition.name = 'rewardCondition' + tier;
          const chestH = Math.min(126, bodyH * .4), chestCenter = bodyH * .39;
          const chest = levelArt(spot, 0, sy(chestCenter), 148, chestH, done ? 'comic_chest_open_v1' : 'comic_chest_closed_v1');
          chest.name = 'rewardChest' + tier;
          if (!done && !canClaim) {
            chest.addComponent(UIOpacity).opacity = 120;
            comicMark(spot, 48, sy(chestCenter + 10), 'lock', 23, COMIC.sub);
          }
          if (done) comicMark(spot, 62, sy(chestCenter + 14), 'check', 24, COMIC.greenShade);
          if (canClaim) {
            const dot = N('rewardReady', spot, 66, sy(chestCenter - chestH / 2 + 7)).addComponent(Graphics);
            dot.fillColor = C('#DF5538'); dot.circle(0, 0, 7); dot.fill();
            dot.strokeColor = C(COMIC.card); dot.lineWidth = 2; dot.stroke();
          }
          const reward: { gold?: number; diamonds?: number } = chestRewards[t];
          const valueTop = chestCenter + chestH / 2 + 25;
          comicArt(spot, -39, sy(valueTop), 28, 28, reward.gold ? 'comic_coin_v1' : 'comic_diamond_v1');
          const amount = mainLabel(spot, 17, sy(valueTop), String(reward.gold || reward.diamonds), {
            size: 24, color: COMIC.text, bold: true, w: 82, h: 30,
          });
          amount.name = 'rewardAmount' + tier;
          const claim = N('rewardStatus' + tier, spot, 0, sy(bodyH - 30), 184, 44);
          if (canClaim) comicSurface(claim, 0, 0, 140, 34, 'gold', { name: 'rewardClaimHint', radius: 17 });
          mainLabel(claim, 0, 0, done ? '已领取' : canClaim ? '一键领取' : '未达成', {
            size: 22, color: done ? COMIC.greenShade : canClaim ? COMIC.text : COMIC.sub, bold: true, w: 172, h: 32,
          });
          if (canClaim) tap(spot, () => {
            if (claiming) return;
            claiming = true;
            const result = claimAllChests(key);
            if (result.ok === false) {
              claiming = false; toast(result.msg, { kind: 'warning', title: '暂时无法领取' }); return;
            }
            ctx.refreshResources?.(); render();
            showRewardPopup(ctx.screens.parent!, lv.id + ' · ' + DIFFS[challenge.difficulty], result.chestCount, [
              { name: '金币', icon: 'comic_coin_v1', amount: result.gold },
              { name: '钻石', icon: 'comic_diamond_v1', amount: result.diamonds },
            ], () => { claiming = false; });
          });
        }
      }

      const restoreBattleEntry = (): void => {
        flow.mode = 'main';
        if (!battleControl.node.isValid) return;
        toast('进入战斗失败，请重试', { kind: 'warning', title: '进入战斗失败' });
      };
      const battleControl = comicButtonControl(content, -91, WY(actionTop, actionH), 490, actionH, '', () => {
        if (!open || enteringBattle()) return;
        if (!campaignDifficultyOpen(lv.id, challenge.difficulty, loadSave().stars)) { toast('请先通关前置关卡或难度'); return; }
        if (!squadList().length) { toast('请先上阵英雄'); return; }
        enterBattle(done => {
          flow.mode = 'battle'; flow.levelId = lv.id; flow.diff = challenge.difficulty;
          return director.loadScene('Main', done);
        }, restoreBattleEntry, warmBattlefield(lv.id, squadList()));
      }, { tone: open ? 'gold' : 'muted', name: 'levelBattle', enabled: open });
      const battle = battleControl.node;
      if (open) {
        levelArt(battle, -161, 0, 62, 62, 'comic_swords_v1');
        mainLabel(battle, 30, 0, '开始战斗', { size: 42, font: 'body', color: COMIC.text, bold: true, w: 318, h: 60 });
      } else {
        comicMark(battle, -178, 0, 'lock', 30, COMIC.sub);
        mainLabel(battle, 20, 0, '通关 ' + LEVELS[st.i - 1].id + ' 解锁', { size: 27, color: COMIC.text, bold: true, w: 400, h: 44 });
      }
      const squad = comicButton(content, 253, WY(actionTop, actionH), 156, actionH, '', () => openSquadPanel(ctx),
        { tone: 'dark', name: 'levelSquad' });
      comicArt(squad, -36, 9, 43, 43, 'comic_helm_falcon_v1');
      mainLabel(squad, 23, 10, '编队', { size: 26, bold: true, color: COMIC.cream, w: 84, h: 36 });
      mainLabel(squad, 0, -27, squadList().length + '/4 已上阵', { size: 18, color: '#C8CFD3', w: 144, h: 26 });
      if (open) {
        const heroes = squadList();
        deferUIWarmup(content, () => { void warmBattlefield(lv.id, heroes); });
      }
    });
  }
  render();
}
