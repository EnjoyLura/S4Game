/** Receipt-only growth celebration. One reusable, non-blocking surface survives page refreshes. */
import { _decorator, Component, game, Game, isValid, Label, Node, UIOpacity, UITransform } from 'cc';
import { LO } from '../config/GameConfig';
import { label, N } from './UIKit';
import { reducedMotion } from './UIMotion';
import { uiSound } from './UIAudio';
import { NoticeStat } from './UINotice';
import { whenUIArtReady } from './UIArtReady';
import { celebrationLight, lightEnvelope, noticeSurface } from './FeedbackSurface';

const { ccclass } = _decorator;
const clamp = (v: number): number => Math.max(0, Math.min(1, v));
const out = (v: number): number => 1 - (1 - clamp(v)) ** 3;
export interface GrowthReceipt {
  key: string; name: string; title: string; level: { before: number; after: number };
  stats: NoticeStat[]; power?: { before: number; after: number }; milestone?: string;
  keep?: boolean; repeating?: boolean;
}
function value(v: number, stat: NoticeStat): string {
  return stat.format === 'percent' ? (Math.round(v * 10000) / 100) + '%'
    : stat.format === 'speed' ? String(Math.round(v * 100) / 100) : Math.round(v).toLocaleString('en-US');
}

@ccclass('PositiveFeedback')
class PositiveFeedback extends Component {
  private receipt: GrowthReceipt | null = null;
  private display: Node | null = null;
  private surface: Node | null = null;
  private surfaceHeight = 0;
  private glimmer!: Node;
  private fade!: UIOpacity;
  private heading!: Label;
  private levels!: Label;
  private power!: Label;
  private stats!: Label;
  private extra!: Label;
  private age = 0;
  private appearanceAge = 0;
  private fromPower = 0;
  private displayedPower = 0;
  private sinceCue = 10;
  private hidden = false;
  private quiet = false;
  private height = 142;
  private lift = 0;
  private last = '';
  onLoad(): void {
    this.enabled = false;
    game.on(Game.EVENT_HIDE, this.hide, this); game.on(Game.EVENT_SHOW, this.show, this);
  }
  present(next: GrowthReceipt): void {
    const previous = this.receipt, merged = previous?.key === next.key && this.display && this.age < 1.8;
    const newMilestone = !!next.milestone;
    this.receipt = merged ? { ...next, milestone: next.milestone || previous!.milestone,
      level: { before: previous!.level.before, after: next.level.after },
      power: next.power && previous!.power ? { before: previous!.power.before, after: next.power.after } : next.power,
      stats: next.stats.map(s => ({ ...s, before: previous!.stats.find(b => b.name === s.name)?.before ?? s.before })) } : next;
    const r = this.receipt;
    if (!merged) { this.appearanceAge = 0; this.displayedPower = r.power?.before || 0; }
    this.fromPower = this.displayedPower;
    this.quiet = reducedMotion(); this.height = r.milestone ? 192 : 142;
    this.lift = Math.min(LO.half * .36, LO.half - LO.notchTop - this.height / 2 - 160);
    if (!this.display) {
      this.display = N('growthCelebration', this.node, 0, this.lift, 680, this.height);
      this.fade = this.display.addComponent(UIOpacity);
      this.glimmer = celebrationLight(this.display, 'growthTitleLight', 0, 40, 500, 96);
      const text = (y: number, size: number, color: string, h = 32): Label =>
        label(this.display!, 0, y, '', { size, color, bold: true, w: 642, h, shrink: true,
          outline: '#142431', outlineW: .8 }).getComponent(Label)!;
      this.heading = text(40, 24, '#FFE4A0'); this.levels = text(0, 22, '#FFF7DE');
      this.levels.node.setPosition(-154, 0, 0); this.levels.node.getComponent(UITransform)!.setContentSize(210, 32);
      this.power = text(0, 31, '#C2EE9F', 42);
      this.power.node.setPosition(98, 0, 0); this.power.node.getComponent(UITransform)!.setContentSize(340, 42);
      this.stats = text(-37, 18, '#E9E3D4'); this.extra = text(-75, 19, '#FFE6A0', 30);
    }
    if (this.surfaceHeight !== this.height) {
      if (this.surface) { this.surface.active = false; this.surface.destroy(); }
      this.surface = noticeSurface(this.display, 680, this.height); this.surface.setSiblingIndex(0);
      this.surfaceHeight = this.height;
    }
    this.heading.string = (r.milestone ? '阶段突破' : r.title) + ' · ' + r.name;
    this.extra.string = r.milestone || '';
    this.last = ''; this.age = 0;
    this.node.setSiblingIndex(this.node.parent!.children.length - 1);
    if (newMilestone || next.keep || !merged || this.sinceCue >= .18) {
      uiSound(next.keep ? 'construction' : newMilestone ? 'breakthrough' : next.repeating ? 'growthTick' : 'upgrade');
      this.sinceCue = 0;
    }
    this.enabled = true; this.render();
  }
  update(dt: number): void {
    if (this.hidden) return;
    const step = Math.min(.06, Math.max(0, dt));
    this.age += step; this.appearanceAge += step; this.sinceCue += step;
    if (this.age >= (this.receipt?.milestone ? 2.45 : 1.85)) { this.clear(); return; }
    this.render();
  }
  private render(): void {
    const r = this.receipt; if (!r || !this.display) return;
    const end = r.milestone ? 2.45 : 1.85, leaving = clamp((this.age - end + .28) / .28);
    this.fade.opacity = Math.round(255 * (this.quiet ? 1 : out(this.appearanceAge / .12)) * (1 - leaving));
    // At appearanceAge zero the screen-space center is y=0, matching ordinary notices.
    const entry = this.lift;
    this.display.setPosition(0, this.lift - (this.quiet ? 0 : entry * (1 - out(this.appearanceAge / .32))) + (this.quiet ? 0 : leaving * 30), 0);
    const progress = this.quiet ? 1 : out(this.age / .45);
    this.levels.string = 'Lv.' + r.level.before + '  →  Lv.' + r.level.after;
    const p = r.power;
    if (p) this.displayedPower = this.fromPower + (p.after - this.fromPower) * progress;
    const gain = p ? Math.round(this.displayedPower - p.before) : 0;
    this.power.string = p && p.after > p.before ? '战力 ' + Math.round(this.displayedPower).toLocaleString('en-US') + '   ↑ +' + gain.toLocaleString('en-US')
      : r.keep ? '防线建设完成' : '成长已生效';
    const stats = r.stats.slice(0, 2).map(s => s.name + ' +' + value(Math.max(0, s.after - s.before), s)).join('     ');
    if (stats !== this.last) { this.stats.string = stats; this.last = stats; }
    lightEnvelope(this.glimmer, this.quiet ? 1 : this.appearanceAge - .24, .7, r.milestone ? .26 : .16);
  }
  private clear(): void {
    if (this.display) { this.display.active = false; this.display.destroy(); }
    this.display = null; this.surface = null; this.surfaceHeight = 0; this.receipt = null; this.enabled = false;
  }
  private hide = (): void => { this.hidden = true; };
  private show = (): void => { this.hidden = false; };
  onDestroy(): void { game.off(Game.EVENT_HIDE, this.hide, this); game.off(Game.EVENT_SHOW, this.show, this); }
}

export function showPositiveFeedback(root: Node, receipt: GrowthReceipt): void {
  if (!isValid(root, true)) return;
  // The action's caller may rebuild its page immediately after returning. Wait for that
  // synchronous rebuild and its opaque draw handoff so the celebration stays in front.
  void Promise.resolve().then(() => whenUIArtReady(root, () => {
    if (!isValid(root, true)) return;
    const host = root.getChildByName('positiveFeedbackLayer') || N('positiveFeedbackLayer', root);
    (host.getComponent(PositiveFeedback) || host.addComponent(PositiveFeedback)).present(receipt);
  }));
}
